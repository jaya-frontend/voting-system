// 1. Initialize the poll variable as a Map object const poll = new Map(); // Initial options with Set objects for votes (incorporating at least 3 default options and votes) poll.set("Turkey", new Set(["traveler1", "traveler2"])); poll.set("Morocco", new Set(["traveler3"])); poll.set("Spain", new Set()); // 2. addOption function function addOption(option) { if (!option || option.trim() === "") { return "Option cannot be empty."; } if (poll.has(option)) { return `Option "${option}" already exists.`; } poll.set(option, new Set()); return `Option "${option}" added to the poll.`; } // 3. vote function function vote(option, voterId) { if (!poll.has(option)) { return `Option "${option}" does not exist.`; } const voters = poll.get(option); if (voters.has(voterId)) { return `Voter ${voterId} has already voted for "${option}".`; } voters.add(voterId); return `Voter ${voterId} voted for "${option}".`; } // 4. displayResults function function displayResults() { let result = "Poll Results:"; for (const [option, voters] of poll) { result += `\n${option}: ${voters.size} votes`; } return result; }
